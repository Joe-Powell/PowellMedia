•Create a Database
CREATE DATABASE DBname;

•SHOW DATABASES
SHOW DATABASES;


•To Delete a Database
DROP DATABASE databasename;


•Step inside a database
Before making any tables or columns you must first step inside the database.
use databasename;


•Create a table and it's columns
CREATE TABLE tablename (
  id int NOT NULL AUTO_INCREMENT,
  url_path VARCHAR(255),
  title VARCHAR(255),
  readmore VARCHAR(255),
  image_url VARCHAR(255),
  category VARCHAR(255),
  created_on TIMESTAMP
  DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id)
  );
  

•Show all tables
SHOW TABLES;


•Delete a table
DROP TABLE table_name;


•Show all columns and rows
SELECT * FROM tablename;


•Rename a table
ALTER TABLE old_tablename RENAME TO new_tablename;


•Add a column
ALTER TABLE tablename ADD Last_name varchar(250);


•Add a column with date
ALTER TABLE tablename ADD date DATE DEFAULT CURRENT_DATE ;


•Add a column with date
ALTER TABLE tablename ADD description varchar(250) AFTER `category`;


•Update/Edit a column
UPDATE tablename SET title = 'newtitle', image_url = 'uploads/newimage_url.png' WHERE id='2';


•Delete a column
ALTER TABLE tablename DROP COLUMN students;


•Insert new record/row
INSERT INTO tablename(username,password,email,created_on) VALUES('mike','763449', 'mikee@mail', CURRENT_TIMESTAMP);


•Delete record/row
DELETE FROM tablename WHERE title = 'sometitle';


•Search for a word in column
SELECT * FROM tablename WHERE searchresults LIKE '%node.js,%';


•Search for exact word, nothing before or after
SELECT * FROM tablename WHERE searchresults LIKE 'node';
